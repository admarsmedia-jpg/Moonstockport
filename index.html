function doGet() {
  const ss    = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('Portfolio');

  const capital  = sheet.getRange('B2').getValue();
  const target   = sheet.getRange('B3').getValue();
  const fxRate   = sheet.getRange('B4').getValue();

  const assets = [];
  for (let row = 7; row <= 11; row++) {
    const name   = sheet.getRange('A' + row).getValue();
    const alloc  = sheet.getRange('B' + row).getValue();
    const yield_ = sheet.getRange('C' + row).getValue();
    const tax    = sheet.getRange('D' + row).getValue();
    const ticker = sheet.getRange('E' + row).getValue();
    if (name) assets.push({ name, alloc, yield: yield_, tax, ticker });
  }

  const payload = {
    capital,
    target,
    fxRate,
    assets,
    updatedAt: new Date().toLocaleString('th-TH', { timeZone: 'Asia/Bangkok' })
  };

  return ContentService
    .createTextOutput(JSON.stringify(payload))
    .setMimeType(ContentService.MimeType.JSON);
}
